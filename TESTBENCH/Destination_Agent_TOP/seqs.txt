class dest_base_seq extends uvm_sequence #(read_xtn);

        `uvm_object_utils(dest_base_seq)

        function new(string name = "dest_base_seq");
                super.new(name);
        endfunction
endclass

class dest_small_seq extends dest_base_seq;

        `uvm_object_utils(dest_small_seq)

        function new(string name = "dest_small_seq");
                super.new(name);
        endfunction

        task body();
           repeat(1)
              begin
                req = read_xtn::type_id::create("req");
                start_item(req);
                assert(req.randomize() with {no_of_cycles inside {[1:28]} ;});
                finish_item(req);
                end
        endtask
endclass


class dest_large_seq extends dest_base_seq;

        `uvm_object_utils(dest_large_seq)

        function new(string name = "dest_large_seq");

        task body();

endclass
